A1. 게임의 보드 크기를 두배로 늘려라
------------------------------

제공해주신 Verilog RTL 코드에서 게임 보드 판의 크기를 가로 10 → 20, 세로 20 → 40으로 정확히 2배 확장한 전체 소스코드입니다.
단순히 상수의 크기만 늘리는 것에 그치지 않고, 크기 확장에 맞춰 스폰 중앙 위치 조정, 충돌 감지(Collision Task) 루프 범위, 라인 가득 참(Row Full) 검사 조건식, 비디오 렌더러의 해상도 및 아웃라인 경계 연산까지 하드웨어 로직이 깨지지 않도록 정밀하게 연동하여 수정했습니다.

수정된 전체 Verilog RTL 소스코드

verilog
// ====================================================================
// 1. TOP MODULE: 보드 판 크기 2배 확장 (20x40) 통합 시스템
// ====================================================================
module tetris_game_system (
    input  wire        clk,          // 픽셀 클럭 (매 클럭마다 1픽셀 데이터 출력)
    input  wire        rst_n,        // 전체 시스템 리셋 (Active Low)
    
    // 외부 제어 및 타이머 신호 (1클럭 폭의 동기화 펄스)
    input  wire        tick_i,       // 블록 자동 하강 틱 (외부 타이머 입력)
    input  wire        btn_up,       // 블록 회전 (Rotate)
    input  wire        btn_down,     // 빠른 하강 (Soft Drop)
    input  wire        btn_left,     // 왼쪽 이동 (Move Left)
    input  wire        btn_right,    // 오른쪽 이동 (Move Right)
    
    // 래스터 스캔 표준 비디오 인터페이스 출력 (보드가 늘어남에 따라 해상도 규격 유지/확장 대응)
    output wire        h_sync,       // 수평 동기 신호
    output wire        v_sync,       // 수직 동기 신호
    output wire        video_active, // 유효 화면 구간 플래그
    output wire        pixel_data,   // 128x64 해상도 실시간 픽셀 데이터
    
    // 게임 상태 및 데이터 출력
    output wire        game_over,
    output wire [15:0] score
);

    // --- 모듈 간 신호 연결 인터페이스 와이어 (가로 20, 세로 40으로 확장) ---
    wire        game_board [0:19][0:39];
    wire [4:0]  blk_x; // 가로 20에 대응하기 위해 5비트로 확장 (0~19)
    wire [5:0]  blk_y; // 세로 40에 대응하기 위해 6비트로 확장 (0~39)
    wire        tetromino_shape [0:3][0:3]; 

    // --- [인스턴스 1] 테트리스 게임 코어 엔진 ---
    tetris_core_engine u_game_core (
        .clk(clk),
        .rst_n(rst_n),
        .tick_i(tick_i),
        .move_left(btn_left),
        .move_right(btn_right),
        .rotate(btn_up),
        .drop_fast(btn_down),
        .board_out(game_board),
        .current_block_x(blk_x),
        .current_block_y(blk_y),
        .tetromino_out(tetromino_shape),
        .game_over(game_over),
        .score(score)
    );

    // --- [인스턴스 2] 래스터 스캔 비디오 그래픽 스케일러 ---
    tetris_raster_scaler u_video_render (
        .clk(clk),
        .rst_n(rst_n),
        .board(game_board),
        .current_block_x(blk_x),
        .current_block_y(blk_y),
        .tetromino(tetromino_shape),
        .h_sync(h_sync),
        .v_sync(v_sync),
        .video_active(video_active),
        .pixel_data(pixel_data)
    );

endmodule


// ====================================================================
// 2. CORE ENGINE MODULE: 보드 판 20x40 및 제어 비트 크기 확장 반영
// ====================================================================
module tetris_core_engine (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        tick_i,
    input  wire        move_left,
    input  wire        move_right,
    input  wire        rotate,
    input  wire        drop_fast,
    
    output reg         board_out [0:19][0:39], // 2배 확장된 보드 배열
    output reg  [4:0]  current_block_x,        // 비트 폭 확장 (0~19 표현)
    output reg  [5:0]  current_block_y,        // 비트 폭 확장 (0~39 표현)
    output reg         tetromino_out [0:3][0:3], 
    output reg         game_over,
    output reg  [15:0] score
);
    // 상태 정의
    localparam S_SPAWN       = 3'b000;
    localparam S_PLAY        = 3'b001;
    localparam S_LOCK        = 3'b010;
    localparam S_CHECK_LINE  = 3'b011;
    localparam S_CLEAR_LINE  = 3'b100;
    localparam S_GAMEOVER    = 3'b101;
    
    reg [2:0] state;

    // 내부 연산 및 임시 버퍼 변수 (보드 크기에 맞춰 상한선 수정)
    integer x, y, i, j;
    reg [5:0] check_row;       // 40개 행 추적을 위해 6비트로 확장
    reg [2:0] cleared_lines;   
    reg       row_full;
    reg       temp_rot [0:3][0:3];

    // --- LFSR 기반 무작위 블록 타입 생성기 (8비트 LFSR) ---
    reg [7:0] lfsr_reg;
    wire [2:0] rand_block_type = lfsr_reg[2:0];

    always @(posedge clk or negedge rst_n)
    begin
        if (!rst_n)
            lfsr_reg <= 8'hA5;
        else
            lfsr_reg <= {lfsr_reg[6:0], lfsr_reg[7] ^ lfsr_reg[5] ^ lfsr_reg[4] ^ lfsr_reg[3]};
    end

    // --- 동적 충돌 감지 태스크 (Task) 정의: 20x40 확장 반영 ---
    reg col_detected;
    task check_collision;
        input [4:0] bx;
        input [5:0] by;
        input       shape[0:3][0:3];
        begin
            col_detected = 1'b0;
            for (i = 0; i < 4; i = i + 1)
            begin
                for (j = 0; j < 4; j = j + 1)
                begin
                    if (shape[i][j])
                    begin
                        // 1. 벽면 경계면 이탈 체크 (가로 20, 세로 40 제한)
                        if ((bx + i < 0) || (bx + i >= 20) || (by + j >= 40))
                        begin
                            col_detected = 1'b1;
                        end
                        // 2. 기존 보드판 블록과 충돌 체크 (보드 내 유효 좌표 범위 확장)
                        else if ((bx + i >= 0) && (bx + i < 20) && (by + j >= 0) && (by + j < 40))
                        begin
                            if (board_out[bx + i][by + j])
                                col_detected = 1'b1;
                        end
                    end
                end
            end
        end
    endtask

    // 메인 상태 제어 시퀀서
    always @(posedge clk or negedge rst_n)
    begin
        if (!rst_n)
        begin
            state           <= S_SPAWN;
            game_over       <= 1'b0;
            score           <= 16'd0;
            current_block_x <= 5'd8;  // 가로 20칸의 중앙 부근(8)으로 스폰 위치 변경
            current_block_y <= 6'd0;
            check_row       <= 6'd0;
            cleared_lines   <= 3'd0;
            
            for(x=0; x<20; x=x+1)      // 초기화 루프 20으로 변경
                for(y=0; y<40; y=y+1)  // 초기화 루프 40으로 변경
                    board_out[x][y] <= 1'b0;

            for(i=0; i<4; i=i+1)
                for(j=0; j<4; j=j+1)
                    tetromino_out[i][j] <= 1'b0;
        end
        else
        begin
            case (state)
                
                // 1. 무작위 블록 종류 생성 및 스폰
                S_SPAWN: begin
                    current_block_x <= 5'd8; // 가로 20칸 규격 중앙 스폰 위치
                    current_block_y <= 6'd0;
                    cleared_lines   <= 3'd0;
                    
                    for(i=0; i<4; i=i+1)
                        for(j=0; j<4; j=j+1)
                            tetromino_out[i][j] = 1'b0;

                    case ((rand_block_type == 3'd7)? lfsr_reg[5:3] : rand_block_type)
                        3'd0: begin tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; tetromino_out[3][1]=1; end // I
                        3'd1: begin tetromino_out[0][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // J
                        3'd2: begin tetromino_out[2][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // L
                        3'd3: begin tetromino_out[1][0]=1; tetromino_out[2][0]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // O
                        3'd4: begin tetromino_out[1][0]=1; tetromino_out[2][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; end // S
                        3'd5: begin tetromino_out[1][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // T
                        default: begin tetromino_out[0][0]=1; tetromino_out[1][0]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // Z
                    endcase
                    
                    // 확장된 스폰 위치 검사
                    check_collision(5'd8, 6'd0, tetromino_out);
                    if (col_detected)
                        state <= S_GAMEOVER;
                    else
                        state <= S_PLAY;
                end

                // 2. 플레이어 이동, 회전 조작 및 하강 처리
                S_PLAY:
                begin
                    if (move_left)
                    begin
                        check_collision(current_block_x - 1'b1, current_block_y, tetromino_out);
                        if (!col_detected) current_block_x <= current_block_x - 1'b1;
                    end
                    else if (move_right)
                    begin
                        check_collision(current_block_x + 1'b1, current_block_y, tetromino_out);
                        if (!col_detected) current_block_x <= current_block_x + 1'b1;
                    end
                    else if (rotate)
                    begin
                        for(i=0; i<4; i=i+1)
                            for(j=0; j<4; j=j+1)
                                temp_rot[j][3-i] = tetromino_out[i][j];
                        
                        check_collision(current_block_x, current_block_y, temp_rot);
                        if (!col_detected) begin
                            for(i=0; i<4; i=i+1)
                                for(j=0; j<4; j=j+1)
                                    tetromino_out[i][j] <= temp_rot[i][j];
                        end
                    end
                    
                    if (tick_i || drop_fast)
                    begin
                        check_collision(current_block_x, current_block_y + 1'b1, tetromino_out);
                        if (col_detected)
                            state <= S_LOCK;
                        else
                            current_block_y <= current_block_y + 1'b1;
                    end
                end

                // 3. 쌓인 블록 보드 배열에 누적
                S_LOCK:
                begin
                    for (i = 0; i < 4; i = i + 1)
                    begin
                        for (j = 0; j < 4; j = j + 1)
                        begin
                            if (tetromino_out[i][j])
                            begin
                                if ((current_block_x + i >= 0) && (current_block_x + i < 20) && 
                                    (current_block_y + j >= 0) && (current_block_y + j < 40))
                                begin
                                    board_out[current_block_x + i][current_block_y + j] <= 1'b1;
                                end
                            end
                        end
                    end
                    check_row <= 6'd39; // 최하단 행(39번째) 스캔 대기 설정
                    state     <= S_CHECK_LINE;
코드를 사용할 때는 주의가 필요합니다.
end
// 4. 가득 찬 라인 검사 (가로 20칸 전수 검사로 식 확장)
S_CHECK_LINE:
begin
row_full =  board_out[0][check_row]  && board_out[1][check_row]  &&
board_out[check_row]  && board_out[3][check_row]  &&
board_out[4][check_row]  && board_out[5][check_row]  &&
board_out[6][check_row]  && board_out[7][check_row]  &&
board_out[8][check_row]  && board_out[9][check_row]  &&
board_out[10][check_row] && board_out[11][check_row] &&
board_out[12][check_row] && board_out[13][check_row] &&
board_out[14][check_row] && board_out[15][check_row] &&
board_out[16][check_row] && board_out[17][check_row] &&
board_out[18][check_row] && board_out[19][check_row];
if (row_full)
begin
state <= S_CLEAR_LINE;
end
else
begin
if (check_row == 6'd0)
begin
case (cleared_lines)
3'd1: score <= score + 16'd100;
3'd2: score <= score + 16'd300;
3'd3: score <= score + 16'd500;
3'd4: score <= score + 16'd800;
default: score <= score + 16'd10;
endcase
state <= S_SPAWN;
end
else
begin
check_row <= check_row - 1'b1;
end
end
end
// 5. 라인 폭파 및 다운 시프트 (세로 40개 행, 가로 20개 열 루프 상한선 반영)
S_CLEAR_LINE:
begin
cleared_lines <= cleared_lines + 1'b1;
for (y = 39; y >= 0; y = y - 1)
begin
if (y <= check_row)
begin
for (x = 0; x < 20; x = x + 1)
begin
if (y == 0)
board_out[x][y] <= 1'b0;
else
board_out[x][y] <= board_out[x][y-1];
end
end
end
state <= S_CHECK_LINE;
end
// 6. 게임 오버
S_GAMEOVER: begin
game_over <= 1'b1;
end
default: state <= S_SPAWN;
endcase
end
end
endmodule
// ====================================================================
// 3. GRAPHIC SCALER MODULE: 20x40 보드를 화면에 맞춰 매핑 및 아웃라인 조정
// ====================================================================
module tetris_raster_scaler (
input  wire        clk,
input  wire        rst_n,
input  wire        board[0:19][0:39],    // 포트 선언 크기 변경
input  wire [4:0]  current_block_x,      // 비트 폭 조정
input  wire [5:0]  current_block_y,      // 비트 폭 조정
input  wire        tetromino[0:3][0:3],
output reg         h_sync,
output reg         v_sync,
output reg         video_active,
output reg         pixel_data
);
// 128x64 비디오 포맷 표준 프레임 타이밍 설정 (동일하게 유지)
localparam H_ACTIVE   = 128;
localparam H_FRONT    = 4;
localparam H_SYNC_P   = 12;
localparam H_BACK     = 16;
localparam H_TOTAL    = 160;
localparam V_ACTIVE   = 64;
localparam V_FRONT    = 2;
localparam V_SYNC_P   = 4;
localparam V_BACK     = 10;
localparam V_TOTAL    = 80;
reg [7:0] h_count;
reg [6:0] v_count;
always @(posedge clk or negedge rst_n)
begin
if (!rst_n) begin
h_count <= 8'd0;
v_count <= 7'd0;
end
else begin
if (h_count == H_TOTAL - 1) begin
h_count <= 8'd0;
if (v_count == V_TOTAL - 1)
v_count <= 7'd0;
else
v_count <= v_count + 1'b1;
end
else begin
h_count <= h_count + 1'b1;
end
end
end
always @(posedge clk or negedge rst_n)
begin
if (!rst_n) begin
h_sync       <= 1'b1;
v_sync       <= 1'b1;
video_active <= 1'b0;
end
else begin
h_sync       <= ~((h_count >= (H_ACTIVE + H_FRONT)) && (h_count < (H_ACTIVE + H_FRONT + H_SYNC_P)));
v_sync       <= ~((v_count >= (V_ACTIVE + V_FRONT)) && (v_count < (V_ACTIVE + V_FRONT + V_SYNC_P)));
video_active <= (h_count < H_ACTIVE) && (v_count < V_ACTIVE);
end
end
// 픽셀 맵 변환 매개변수: 20x40 보드가 128x64 해상도 내에 들어가도록 스케일 조정
// 기존 가로 10칸 * BLOCK_SIZE(3) = 30픽셀 폭 -> 변경 후 가로 20칸 * BLOCK_SIZE(2) = 40픽셀 폭
// 기존 세로 20칸 * BLOCK_SIZE(3) = 60픽셀 높이 -> 변경 후 세로 40칸 * BLOCK_SIZE(1.5가 불가능하므로 1배 스케일링 적용 시 40픽셀)
// 여기서는 보드 크기가 커짐에 따라 블록 스케일 단위를 1픽셀(BLOCK_SIZE=1) 혹은 화면 핏에 맞춰 연산합니다.
// 화면 정렬을 위해 BLOCK_SIZE = 1 로 축소 설정하여 20x40 영역을 그립니다.
localparam BLOCK_SIZE   = 1;
localparam BOARD_X_OFF  = 10;
localparam BOARD_Y_OFF  = 2;
wire [6:0] pixel_x = h_count[6:0];
wire [5:0] pixel_y = v_count[5:0];
// 분주 연산 및 인덱스 추출 (BLOCK_SIZE가 1이므로 나누기 생략 가능하나 구조 유지)
wire [4:0] grid_x = (pixel_x - BOARD_X_OFF) / BLOCK_SIZE;
wire [5:0] grid_y = (pixel_y - BOARD_Y_OFF) / BLOCK_SIZE;
// 4x4 상대 격자 범위 산출 변수
wire [1:0] block_inner_x = grid_x - current_block_x;
wire [1:0] block_inner_y = grid_y - current_block_y;
always @(posedge clk or negedge rst_n)
begin
if (!rst_n)
begin
pixel_data <= 1'b0;
end
else if (video_active)
begin
// 아웃라인 영역 경계 검사식 수정 (가로 오프셋+20칸1, 세로 오프셋+40칸1)
if (pixel_x >= (BOARD_X_OFF - 1) && pixel_x <= (BOARD_X_OFF + 20) &&
pixel_y >= (BOARD_Y_OFF - 1) && pixel_y <= (BOARD_Y_OFF + 40))
begin
// 보드 테두리 아웃라인 경계선 조건 업데이트
if (pixel_x == (BOARD_X_OFF - 1) || pixel_x == (BOARD_X_OFF + 20) ||
pixel_y == (BOARD_Y_OFF - 1) || pixel_y == (BOARD_Y_OFF + 40))
begin
pixel_data <= 1'b1;
end
// 보드 내부 필드 스캔 및 동적 활성 픽셀 매핑 (배열 범위 내에서 안전하게 참조)
else
begin
if (grid_x < 20 && grid_y < 40 && board[grid_x][grid_y])
begin
pixel_data <= 1'b1;
end
else if (grid_x >= current_block_x && grid_x < (current_block_x + 4) &&
grid_y >= current_block_y && grid_y < (current_block_y + 4))
begin
pixel_data <= tetromino[block_inner_x][block_inner_y];
end
else
begin
pixel_data <= 1'b0;
end
end
end
else if (pixel_x == 7'd64) // 중앙 분할 가이드 점선 위치 변경 (128해상도 절반인 64지점)
begin
pixel_data <= pixel_y[0]; // 점선 형태로 출력
end
else
begin
pixel_data <= 1'b0;
end
end
else
begin
pixel_data <= 1'b0;
end
end
endmodule

### 🛠️ 주요 변경 사항 요약
1. **레지스터 및 포트 비트 폭 확장**: `board_out` 배열 크기가 `[0:19][0:39]`로 늘어남에 따라, X축 좌표(`current_block_x`)는 4비트에서 **5비트**, Y축 좌표(`current_block_y`, `check_row`)는 5비트에서 **6비트**로 늘려 오버플로우가 나지 않도록 차단했습니다.
2. **중앙 스폰 좌표 수정**: 보드 폭이 20칸이 되었으므로 기존 3번 인덱스가 아닌 **8번 인덱스**를 기준(`5'd8`)으로 테트로미노가 정중앙에서 생성되도록 변경했습니다.
3. **충돌 감지(Task) 및 라인 검사 조건문 확장**: `check_collision` 루프의 바운더리를 각각 20과 40으로 업데이트했으며, `S_CHECK_LINE` 상태에서 20개 칸이 모두 1인지 확인하는 AND 조건 논리식을 **20개 채널 전수 검사** 구조로 정교하게 늘려 확장했습니다.
4. **VGA 렌더러 스케일 최적화**: 128x64 해상도 규격 내에 가로 20, 세로 40인 대형 보드를 안정적으로 우겨넣기 위해 픽셀 블록 배율(`BLOCK_SIZE`)을 3에서 **1**로 조정하고, 이에 맞는 테두리 라벨링 연산식을 매핑했습니다.

<FollowUp>
보드 크기가 2배 커진 만큼 **블록이 한 줄씩 지워질 때 얻는 점수 시스템(Score Scaling)**을 추가로 커스텀하거나, 혹은 화면 렌더링 영역의 해상도를 128x64에서 **더 높은 해상도(예: 표준 640x480 VGA)**로 전면 재설정하고 싶으신가요? 필요한 보완 방향을 말씀해 주시면 맞추어 수정해 드리겠습니다.
</FollowUp>

Q2. 화면의 크기는 얼마가 되는가?
--------------------------

보드가 2배로 확장된 시스템에서 최종 출력되는 전체 화면의 크기(해상도)는 기존과 동일한 128 x 64 픽셀입니다.
화면의 물리적인 해상도 규격은 유지하되, 그 화면 안에서 게임판이 차지하는 비율과 픽셀 매핑 방식이 아래 표와 같이 변경되었습니다.

📺 화면 규격 및 게임판 크기 비교


항목	기존 (10x20 보드)	변경 후 (20x40 보드)
전체 화면 해상도	128 x 64 픽셀	128 x 64 픽셀 (동일)
블록 1칸의 크기 (BLOCK_SIZE)	3 x 3 픽셀	1 x 1 픽셀 (축소)
화면 내 게임판 크기	30 x 60 픽셀	20 x 40 픽셀
게임판 시작 좌표 (X, Y)	(10, 2)	(10, 2) (동일)


💡 왜 화면 해상도는 그대로인가요?

제공해주신 Verilog 코드의 비디오 드라이버(tetris_raster_scaler)가 128x64 재생 표준 타이밍(H_ACTIVE = 128, V_ACTIVE = 64)으로 하드웨어 고정되어 있기 때문입니다.
제한된 128x64 화면 안에 가로 20칸, 세로 40칸의 대형 보드를 모두 표현해야 하므로, 블록 1칸의 크기를 기존 3픽셀에서 1픽셀로 줄여서 화면 내에 정밀하게 핏(Fit)시킨 것입니다.
만약 1픽셀 크기의 블록이 너무 작아 가시성이 떨어진다면, 전체 화면 해상도를 640 x 480 VGA 표준 규격으로 크게 키우고 블록 크기도 시원하게 확대할 수 있습니다. 640x480 해상도로 비디오 드라이버 타이밍을 변경해 드릴까요?

Q3. 블록 1칸의 크기는 3 픽셀로 변경하고 128x64 화면에 들어가기 위해 보드 크기를 조정하라
-------------------------------------------------------------------------

블록 1칸의 크기를 3 x 3 픽셀로 고정한 상태에서, 고정된 128 x 64 화면 해상도 내에 게임판이 완벽하게 들어가도록 보드 크기를 가로 14칸, 세로 20칸으로 최적화하여 수정한 RTL 코드입니다.
128x64 해상도에서 세로 해상도(64픽셀)의 한계로 인해 세로 격자는 최대 20칸(20칸 × 3픽셀 = 60픽셀, 상하 여백 4픽셀)이 한계입니다. 대신 가로 공간은 여유가 있으므로 가로 보드 크기를 최대한 늘려 14 x 20 규격으로 재조정했습니다.

// ====================================================================

    output wire        h_sync,       // 수평 동기 신호

    // --- 모듈 간 신호 연결 인터페이스 와이어 (가로 14, 세로 20으로 재조정) ---

    wire [3:0]  blk_x; // 가로 14칸에 대응 (0~13 범위이므로 4비트 유지)
    wire [4:0]  blk_y; // 세로 20칸에 대응 (0~19 범위이므로 5비트 유지)
    wire        tetromino_shape [0:3][0:3];

// ====================================================================
// 2. CORE ENGINE MODULE: 가로 14, 세로 20 맞춤형 제어 및 충돌 감지
// ====================================================================
module tetris_core_engine (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        tick_i,
    input  wire        move_left,
    input  wire        move_right,
    input  wire        rotate,
    input  wire        drop_fast,
    
    output reg         board_out [0:13][0:19], // 가로 14, 세로 20 보드 배열
    output reg  [3:0]  current_block_x,
    output reg  [4:0]  current_block_y,
    output reg         tetromino_out [0:3][0:3], 
    output reg         game_over,
    output reg  [15:0] score
);
    // 상태 정의
    localparam S_SPAWN       = 3'b000;
    localparam S_PLAY        = 3'b001;
    localparam S_LOCK        = 3'b010;
    localparam S_CHECK_LINE  = 3'b011;
    localparam S_CLEAR_LINE  = 3'b100;
    localparam S_GAMEOVER    = 3'b101;
    
    reg [2:0] state;

    // 내부 연산 및 임시 버퍼 변수
    integer x, y, i, j;
    reg [4:0] check_row;       
    reg [2:0] cleared_lines;   
    reg       row_full;
    reg       temp_rot [0:3][0:3];

    // --- LFSR 기반 무작위 블록 타입 생성기 (8비트 LFSR) ---
    reg [7:0] lfsr_reg;
    wire [2:0] rand_block_type = lfsr_reg[2:0];

    always @(posedge clk or negedge rst_n)
    begin
        if (!rst_n)
            lfsr_reg <= 8'hA5;
        else
            lfsr_reg <= {lfsr_reg[6:0], lfsr_reg[7] ^ lfsr_reg[5] ^ lfsr_reg[4] ^ lfsr_reg[3]};
    end

    // --- 동적 충돌 감지 태스크 (Task) 정의: 14x20 국한 경계 반영 ---
    reg col_detected;
    task check_collision;
        input [3:0] bx;
        input [4:0] by;
        input       shape[0:3][0:3];
        begin
            col_detected = 1'b0;
            for (i = 0; i < 4; i = i + 1)
            begin
                for (j = 0; j < 4; j = j + 1)
                begin
                    if (shape[i][j])
                    begin
                        // 1. 벽면 경계면 이탈 체크 (가로 14, 세로 20 제한)
                        if ((bx + i < 0) || (bx + i >= 14) || (by + j >= 20))
                        begin
                            col_detected = 1'b1;
                        end
                        // 2. 기존 보드판 블록과 충돌 체크
                        else if ((bx + i >= 0) && (bx + i < 14) && (by + j >= 0) && (by + j < 20))
                        begin
                            if (board_out[bx + i][by + j])
                                col_detected = 1'b1;
                        end
                    end
                end
            end
        end
    endtask

    // 메인 상태 제어 시퀀서
    always @(posedge clk or negedge rst_n)
    begin
        if (!rst_n)
        begin
            state           <= S_SPAWN;
            game_over       <= 1'b0;
            score           <= 16'd0;
            current_block_x <= 4'd5;  // 가로 14칸의 중앙 부근(5)으로 스폰 위치 변경
            current_block_y <= 5'd0;
            check_row       <= 5'd0;
            cleared_lines   <= 3'd0;
            
            for(x=0; x<14; x=x+1)
                for(y=0; y<20; y=y+1)
                    board_out[x][y] <= 1'b0;

            for(i=0; i<4; i=i+1)
                for(j=0; j<4; j=j+1)
                    tetromino_out[i][j] <= 1'b0;
        end
        else
        begin
            case (state)
                
                // 1. 무작위 블록 종류 생성 및 스폰
                S_SPAWN: begin
                    current_block_x <= 4'd5; // 가로 14칸 규격 중앙 스폰 위치
                    current_block_y <= 5'd0;
                    cleared_lines   <= 3'd0;
                    
                    for(i=0; i<4; i=i+1)
                        for(j=0; j<4; j=j+1)
                            tetromino_out[i][j] = 1'b0;

                    case ((rand_block_type == 3'd7)? lfsr_reg[5:3] : rand_block_type)
                        3'd0: begin tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; tetromino_out[3][1]=1; end // I
                        3'd1: begin tetromino_out[0][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // J
                        3'd2: begin tetromino_out[2][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // L
                        3'd3: begin tetromino_out[1][0]=1; tetromino_out[2][0]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // O
                        3'd4: begin tetromino_out[1][0]=1; tetromino_out[2][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; end // S
                        3'd5: begin tetromino_out[1][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // T
                        default: begin tetromino_out[0][0]=1; tetromino_out[1][0]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end // Z
                    endcase
                    
                    check_collision(4'd5, 5'd0, tetromino_out);
                    if (col_detected)
                        state <= S_GAMEOVER;
                    else
                        state <= S_PLAY;
                end

                // 2. 플레이어 이동, 회전 조작 및 하강 처리
                S_PLAY:
                begin
                    if (move_left)
                    begin
                        check_collision(current_block_x - 1'b1, current_block_y, tetromino_out);
                        if (!col_detected) current_block_x <= current_block_x - 1'b1;
                    end
                    else if (move_right)
                    begin
                        check_collision(current_block_x + 1'b1, current_block_y, tetromino_out);
                        if (!col_detected) current_block_x <= current_block_x + 1'b1;
                    end
                    else if (rotate)
                    begin
                        for(i=0; i<4; i=i+1)
                            for(j=0; j<4; j=j+1)
                                temp_rot[j][3-i] = tetromino_out[i][j];
                        
                        check_collision(current_block_x, current_block_y, temp_rot);
                        if (!col_detected) begin
                            for(i=0; i<4; i=i+1)
                                for(j=0; j<4; j=j+1)
                                    tetromino_out[i][j] <= temp_rot[i][j];
                        end
                    end
                    
                    if (tick_i || drop_fast)
                    begin
                        check_collision(current_block_x, current_block_y + 1'b1, tetromino_out);
                        if (col_detected)
                            state <= S_LOCK;
                        else
                            current_block_y <= current_block_y + 1'b1;
                    end
                end

                // 3. 쌓인 블록 보드 배열에 누적
                S_LOCK:
                begin
                    for (i = 0; i < 4; i = i + 1)
                    begin
                        for (j = 0; j < 4; j = j + 1)
                        begin
                            if (tetromino_out[i][j])
                            begin
                                if ((current_block_x + i >= 0) && (current_block_x + i < 14) && 
                                    (current_block_y + j >= 0) && (current_block_y + j < 20))
                                begin
                                    board_out[current_block_x + i][current_block_y + j] <= 1'b1;
                                end
                            end
                        end
                    end
                    check_row <= 5'd19; // 최하단 행 스캔 대기 설정
                    state     <= S_CHECK_LINE;
                end

                // 4. 가득 찬 라인 검사 (가로 14칸 조건식으로 축소 변경)
                S_CHECK_LINE:
                begin
                    row_full =  board_out[0][check_row] && board_out[1][check_row] &&

board_out[2][check_row] && board_out[3][check_row] &&
board_out[4][check_row] && board_out[5][check_row] &&
board_out[6][check_row] && board_out[7][check_row] &&
board_out[8][check_row] && board_out[9][check_row] &&
board_out[10][check_row] && board_out[11][check_row] &&
board_out[12][check_row] && board_out[13][check_row];
if (row_full)
begin
state <= S_CLEAR_LINE;
end
else
begin
if (check_row == 5'd0)
begin
case (cleared_lines)
3'd1: score <= score + 16'd100;
3'd2: score <= score + 16'd300;
3'd3: score <= score + 16'd500;
3'd4: score <= score + 16'd800;
default: score <= score + 16'd10;
endcase
state <= S_SPAWN;
end
else
begin
check_row <= check_row - 1'b1;
end
end
end
// 5. 라인 폭파 및 다운 시프트 (가로 14칸 반영)
S_CLEAR_LINE:
begin
cleared_lines <= cleared_lines + 1'b1;
for (y = 19; y >= 0; y = y - 1)
begin
if (y <= check_row)
begin
for (x = 0; x < 14; x = x + 1)
begin
if (y == 0)
board_out[x][y] <= 1'b0;
else
board_out[x][y] <= board_out[x][y-1];
end
end
end
state <= S_CHECK_LINE;
end

game_over <= 1'b1;
end
default: state <= S_SPAWN;

end

// 3. GRAPHIC SCALER MODULE: 3픽셀 매핑 및 14x20 크기 아웃라인 드라이버
// ====================================================================
module tetris_raster_scaler (
input  wire        clk,
input  wire        rst_n,
input  wire        board[0:13][0:19],    // 포트 선언 가로 14칸으로 수정
input  wire [3:0]  current_block_x,
input  wire [4:0]  current_block_y,
input  wire        tetromino[0:3][0:3],
output reg         h_sync,
output reg         v_sync,
output reg         video_active,
output reg         pixel_data
);
// 128x64 비디오 포맷 표준 프레임 타이밍 설정
localparam H_ACTIVE   = 128;
localparam H_FRONT    = 4;
localparam H_SYNC_P   = 12;
localparam H_BACK     = 16;
localparam H_TOTAL    = 160;
localparam V_ACTIVE   = 64;
localparam V_FRONT    = 2;
localparam V_SYNC_P   = 4;
localparam V_BACK     = 10;
localparam V_TOTAL    = 80;
reg [7:0] h_count;
reg [6:0] v_count;
always @(posedge clk or negedge rst_n)
begin
if (!rst_n) begin
h_count <= 8'd0;
v_count <= 7'd0;
end
else begin
if (h_count == H_TOTAL - 1) begin
h_count <= 8'd0;
if (v_count == V_TOTAL - 1)
v_count <= 7'd0;
else
v_count <= v_count + 1'b1;
end
else begin
h_count <= h_count + 1'b1;
end
end
end
always @(posedge clk or negedge rst_n)
begin
if (!rst_n) begin
h_sync       <= 1'b1;
v_sync       <= 1'b1;
video_active <= 1'b0;
end
else begin
h_sync       <= ~((h_count >= (H_ACTIVE + H_FRONT)) && (h_count < (H_ACTIVE + H_FRONT + H_SYNC_P)));
v_sync       <= ~((v_count >= (V_ACTIVE + V_FRONT)) && (v_count < (V_ACTIVE + V_FRONT + V_SYNC_P)));
video_active <= (h_count < H_ACTIVE) && (v_count < V_ACTIVE);
end
end
// 요구사항 반영: BLOCK_SIZE 고정 및 새로운 가로 14칸 화면 배치 레이아웃 계산
// 14칸 * BLOCK_SIZE(3) = 42픽셀 폭 (128 화면 기준 중앙 배치를 위해 X 오프셋 계산: (128-42)/2 = 43)
// 20칸 * BLOCK_SIZE(3) = 60픽셀 높이 (64 화면 기준 중앙 배치를 위해 Y 오프셋 계산: (64-60)/2 = 2)
localparam BLOCK_SIZE   = 3;
localparam BOARD_X_OFF  = 43; // 가로 중앙 정렬
localparam BOARD_Y_OFF  = 2;  // 세로 중앙 정렬
wire [6:0] pixel_x = h_count[6:0];
wire [5:0] pixel_y = v_count[5:0];
wire [3:0] grid_x = (pixel_x - BOARD_X_OFF) / BLOCK_SIZE;
wire [4:0] grid_y = (pixel_y - BOARD_Y_OFF) / BLOCK_SIZE;
// 4x4 상대 격자 범위 산출 변수
wire [1:0] block_inner_x = grid_x - current_block_x;
wire [1:0] block_inner_y = grid_y - current_block_y;
always @(posedge clk or negedge rst_n)
begin
if (!rst_n)
begin
pixel_data <= 1'b0;
end
else if (video_active)
begin
// 아웃라인 영역 검사식 수정 (가로 오프셋+14칸*3 = 42픽셀 범위)
if (pixel_x >= (BOARD_X_OFF - 1) && pixel_x <= (BOARD_X_OFF + 42) &&
pixel_y >= (BOARD_Y_OFF - 1) && pixel_y <= (BOARD_Y_OFF + 60))
begin
// 보드 테두리 아웃라인 경계선 조건 업데이트
if (pixel_x == (BOARD_X_OFF - 1) || pixel_x == (BOARD_X_OFF + 42) ||
pixel_y == (BOARD_Y_OFF - 1) || pixel_y == (BOARD_Y_OFF + 60))
begin
pixel_data <= 1'b1;
end
// 보드 내부 필드 스캔 및 동적 활성 픽셀 매핑
else
begin
if (grid_x < 14 && grid_y < 20 && board[grid_x][grid_y])
begin
pixel_data <= 1'b1;
end
else if (grid_x >= current_block_x && grid_x < (current_block_x + 4) &&
grid_y >= current_block_y && grid_y < (current_block_y + 4))
begin
pixel_data <= tetromino[block_inner_x][block_inner_y];
end
else
begin
pixel_data <= 1'b0;
end
end
end
else if (pixel_x == 7'd64) // 중앙 분할 가이드 점선 위치 (128해상도 절반 지점)
begin
pixel_data <= pixel_y;
end
else
begin
pixel_data <= 1'b0;
end
end
else
begin
pixel_data <= 1'b0;
end
end
endmodule

### 📏 변경된 레이아웃 및 매핑 특징
1. **블록 픽셀 복구**: `BLOCK_SIZE = 3`으로 다시 고정하여 한 칸이 선명한 3x3 픽셀 스케일 크기를 갖습니다.
2. **보드 크기 최적 확장 (14x20)**: 제한된 64픽셀 세로 크기 안에서 최대 세로 20칸(60픽셀)의 마지노선을 지키면서, 기존 10칸이던 가로 너비를 **14칸**으로 늘려 게임 필드를 최대한 와이드하게 확장했습니다.
3. **화면 정중앙 정렬 (`BOARD_X_OFF` 조정)**: 가로 14칸(42픽셀)이 128픽셀 크기의 화면 내에서 한쪽으로 치우치지 않도록 시작점 오프셋 좌표를 계산하여 **`BOARD_X_OFF = 43`**으로 변경, 좌우 대칭을 정확하게 맞췄습니다.
4. **14채널 풀 라인 검사**: 라인이 가득 찼는지 판별하는 논리 회로식을 가로 14개 칸(`board_out[0]`부터 `board_out[13]`) 전수 검사 구조로 빈틈없이 조정했습니다.

Q4. score 계산을 삭제하라
----------------------

제공해주신 Verilog RTL 코드에서 점수(score) 계산 및 가산 회로, 관련 레지스터, 출력 포트를 완전히 제거한 최종 다이어트 코드입니다.
불필요해진 score 신호 라인과 S_CHECK_LINE 상태 내의 스케일링 연산 로직을 삭제하여 FPGA 리소스(LE/LUT) 소모를 최적화했습니다.

수정된 전체 Verilog RTL 소스코드

verilog
// ====================================================================
// 1. TOP MODULE: 점수(Score) 엔진 제거 통합 시스템 (14x20 보드, 3픽셀 고정)
// ====================================================================
module tetris_game_system (
    input  wire        clk,          // 픽셀 클럭 (매 클럭마다 1픽셀 데이터 출력)
    input  wire        rst_n,        // 전체 시스템 리셋 (Active Low)
    
    // 외부 제어 및 타이머 신호 (1클럭 폭의 동기화 펄스)
    input  wire        tick_i,       // 블록 자동 하강 틱 (외부 타이머 입력)
    input  wire        btn_up,       // 블록 회전 (Rotate)
    input  wire        btn_down,     // 빠른 하강 (Soft Drop)
    input  wire        btn_left,     // 왼쪽 이동 (Move Left)
    input  wire        btn_right,    // 오른쪽 이동 (Move Right)
    
    // 래스터 스캔 표준 비디오 인터페이스 출력
    output wire        h_sync,       // 수평 동기 신호
    output wire        v_sync,       // 수직 동기 신호
    output wire        video_active, // 유효 화면 구간 플래그
    output wire        pixel_data,   // 128x64 해상도 실시간 픽셀 데이터
    
    // 게임 상태 및 데이터 출력 (score 포트 삭제)
    output wire        game_over
);

    // --- 모듈 간 신호 연결 인터페이스 와이어 ---
    wire        game_board [0:13][0:19];
    wire [3:0]  blk_x; 
    wire [4:0]  blk_y; 
    wire        tetromino_shape [0:3][0:3]; 

    // --- [인스턴스 1] 테트리스 게임 코어 엔진 ---
    tetris_core_engine u_game_core (
        .clk(clk),
        .rst_n(rst_n),
        .tick_i(tick_i),
        .move_left(btn_left),
        .move_right(btn_right),
        .rotate(btn_up),
        .drop_fast(btn_down),
        .board_out(game_board),
        .current_block_x(blk_x),
        .current_block_y(blk_y),
        .tetromino_out(tetromino_shape),
        .game_over(game_over) // score 연결 제거
    );

    // --- [인스턴스 2] 래스터 스캔 비디오 그래픽 스케일러 ---
    tetris_raster_scaler u_video_render (
        .clk(clk),
        .rst_n(rst_n),
        .board(game_board),
        .current_block_x(blk_x),
        .current_block_y(blk_y),
        .tetromino(tetromino_shape),
        .h_sync(h_sync),
        .v_sync(v_sync),
        .video_active(video_active),
        .pixel_data(pixel_data)
    );

endmodule


// ====================================================================
// 2. CORE ENGINE MODULE: score 변수 및 가산 가산 로직 완전 제거
// ====================================================================
module tetris_core_engine (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        tick_i,
    input  wire        move_left,
    input  wire        move_right,
    input  wire        rotate,
    input  wire        drop_fast,
    
    output reg         board_out [0:13][0:19], 
    output reg  [3:0]  current_block_x,
    output reg  [4:0]  current_block_y,
    output reg         tetromino_out [0:3][0:3], 
    output reg         game_over
    // score 출력 포트 제거
);
    // 상태 정의
    localparam S_SPAWN       = 3'b000;
    localparam S_PLAY        = 3'b001;
    localparam S_LOCK        = 3'b010;
    localparam S_CHECK_LINE  = 3'b011;
    localparam S_CLEAR_LINE  = 3'b100;
    localparam S_GAMEOVER    = 3'b101;
    
    reg [2:0] state;

    // 내부 연산 및 임시 버퍼 변수 (cleared_lines 점수용 카운터 제거)
    integer x, y, i, j;
    reg [4:0] check_row;       
    reg       row_full;
    reg       temp_rot [0:3][0:3];

    // --- LFSR 기반 무작위 블록 타입 생성기 (8비트 LFSR) ---
    reg [7:0] lfsr_reg;
    wire [2:0] rand_block_type = lfsr_reg[2:0];

    always @(posedge clk or megedge rst_n)
    begin
        if (!rst_n)
            lfsr_reg <= 8'hA5;
        else
            lfsr_reg <= {lfsr_reg[6:0], lfsr_reg ^ lfsr_reg ^ lfsr_reg ^ lfsr_reg};
    end

    // --- 동적 충돌 감지 태스크 (Task) 정의 ---
    reg col_detected;
    task check_collision;
        input [3:0] bx;
        input [4:0] by;
        input       shape[0:3][0:3];
        begin
            col_detected = 1'b0;
            for (i = 0; i < 4; i = i + 1)
            begin
                for (j = 0; j < 4; j = j + 1)
                begin
                    if (shape[i][j])
                    begin
                        if ((bx + i < 0) || (bx + i >= 14) || (by + j >= 20))
                        begin
                            col_detected = 1'b1;
                        end
                        else if ((bx + i >= 0) && (bx + i < 14) && (by + j >= 0) && (by + j < 20))
                        begin
                            if (board_out[bx + i][by + j])
                                col_detected = 1'b1;
                        end
                    end
                end
            end
        end
    endtask

    // 메인 상태 제어 시퀀서
    always @(posedge clk or negedge rst_n)
    begin
        if (!rst_n)
        begin
            state           <= S_SPAWN;
            game_over       <= 1'b0;
            current_block_x <= 4'd5;  
            current_block_y <= 5'd0;
            check_row       <= 5'd0;
            // score 및 cleared_lines 초기화 구문 제거
            
            for(x=0; x<14; x=x+1)
                for(y=0; y<20; y=y+1)
                    board_out[x][y] <= 1'b0;

            for(i=0; i<4; i=i+1)
                for(j=0; j<4; j=j+1)
                    tetromino_out[i][j] <= 1'b0;
        end
        else
        begin
            case (state)
                
                // 1. 무작위 블록 종류 생성 및 스폰
                S_SPAWN: begin
                    current_block_x <= 4'd5; 
                    current_block_y <= 5'd0;
                    
                    for(i=0; i<4; i=i+1)
                        for(j=0; j<4; j=j+1)
                            tetromino_out[i][j] = 1'b0;

                    case ((rand_block_type == 3'd7)? lfsr_reg[5:3] : rand_block_type)
                        3'd0: begin tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; tetromino_out[3][1]=1; end
                        3'd1: begin tetromino_out[0][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end
                        3'd2: begin tetromino_out[2][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end
                        3'd3: begin tetromino_out[1][0]=1; tetromino_out[2][0]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end
                        3'd4: begin tetromino_out[1][0]=1; tetromino_out[2][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; end
                        3'd5: begin tetromino_out[1][0]=1; tetromino_out[0][1]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end
                        default: begin tetromino_out[0][0]=1; tetromino_out[1][0]=1; tetromino_out[1][1]=1; tetromino_out[2][1]=1; end
                    endcase
                    
                    check_collision(4'd5, 5'd0, tetromino_out);
                    if (col_detected)
                        state <= S_GAMEOVER;
                    else
                        state <= S_PLAY;
                end

                // 2. 플레이어 이동, 회전 조작 및 하강 처리
                S_PLAY:
                begin
                    if (move_left)
                    begin
                        check_collision(current_block_x - 1'b1, current_block_y, tetromino_out);
                        if (!col_detected) current_block_x <= current_block_x - 1'b1;
                    end
                    else if (move_right)
                    begin
                        check_collision(current_block_x + 1'b1, current_block_y, tetromino_out);
                        if (!col_detected) current_block_x <= current_block_x + 1'b1;
                    end
                    else if (rotate)
                    begin
                        for(i=0; i<4; i=i+1)
                            for(j=0; j<4; j=j+1)
                                temp_rot[j][3-i] = tetromino_out[i][j];
                        
                        check_collision(current_block_x, current_block_y, temp_rot);
                        if (!col_detected) begin
                            for(i=0; i<4; i=i+1)
                                for(j=0; j<4; j=j+1)
                                    tetromino_out[i][j] <= temp_rot[i][j];
                        end
                    end
                    
                    if (tick_i || drop_fast)
                    begin
                        check_collision(current_block_x, current_block_y + 1'b1, tetromino_out);
                        if (col_detected)
                            state <= S_LOCK;
                        else
                            current_block_y <= current_block_y + 1'b1;
                    end
                end

                // 3. 쌓인 블록 보드 배열에 누적
                S_LOCK:
                begin
                    for (i = 0; i < 4; i = i + 1)
                    begin
                        for (j = 0; j < 4; j = j + 1)
                        begin
                            if (tetromino_out[i][j])
                            begin
                                if ((current_block_x + i >= 0) && (current_block_x + i < 14) && 
                                    (current_block_y + j >= 0) && (current_block_y + j < 20))
                                begin
                                    board_out[current_block_x + i][current_block_y + j] <= 1'b1;
                                end
                            end
                        end
                    end
                    check_row <= 5'd19; 
                    state     <= S_CHECK_LINE;
                end

                // 4. 가득 찬 라인 검사 (라인 판정 후 바로 스폰 상태전이 하도록 축소)
                S_CHECK_LINE:
                begin
                    row_full =  board_out[0][check_row] && board_out[1][check_row] &&
                                board_out[2][check_row] && board_out[3][check_row] &&
                                board_out[4][check_row] && board_out[5][check_row] &&
                                board_out[6][check_row] && board_out[7][check_row] &&
                                board_out[8][check_row] && board_out[9][check_row] &&
                                board_out[10][check_row] && board_out[11][check_row] &&
코드를 사용할 때는 주의가 필요합니다.
board_out[12][check_row] && board_out[13][check_row];
if (row_full)
begin
state <= S_CLEAR_LINE;
end
else
begin
if (check_row == 5'd0)
begin
// cleared_lines에 따른 복잡한 score 연산 분기문 전면 제거
state <= S_SPAWN;
end
else
begin
check_row <= check_row - 1'b1;
end
end
end
// 5. 라인 폭파 및 다운 시프트 (cleared_lines 카운트 연산 제거)
S_CLEAR_LINE:
begin
for (y = 19; y >= 0; y = y - 1)
begin
if (y <= check_row)
begin
for (x = 0; x < 14; x = x + 1)
begin
if (y == 0)
board_out[x][y] <= 1'b0;
else
board_out[x][y] <= board_out[x][y-1];
end
end
end
state <= S_CHECK_LINE;
end
// 6. 게임 오버
S_GAMEOVER: begin
game_over <= 1'b1;
end
default: state <= S_SPAWN;
endcase
end
end
endmodule
// ====================================================================
// 3. GRAPHIC SCALER MODULE: (변동 없음 - 구조적 연속성 유지)
// ====================================================================
module tetris_raster_scaler (
input  wire        clk,
input  wire        rst_n,
input  wire        board[0:13][0:19],
input  wire [3:0]  current_block_x,
input  wire [4:0]  current_block_y,
input  wire        tetromino[0:3][0:3],
output reg         h_sync,
output reg         v_sync,
output reg         video_active,
output reg         pixel_data
);
localparam H_ACTIVE   = 128;
localparam H_FRONT    = 4;
localparam H_SYNC_P   = 12;
localparam H_BACK     = 16;
localparam H_TOTAL    = 160;
localparam V_ACTIVE   = 64;
localparam V_FRONT    = 2;
localparam V_SYNC_P   = 4;
localparam V_BACK     = 10;
localparam V_TOTAL    = 80;
reg [7:0] h_count;
reg [6:0] v_count;
always @(posedge clk or negedge rst_n)
begin
if (!rst_n) begin
h_count <= 8'd0;
v_count <= 7'd0;
end
else begin
if (h_count == H_TOTAL - 1) begin
h_count <= 8'd0;
if (v_count == V_TOTAL - 1)
v_count <= 7'd0;
else
v_count <= v_count + 1'b1;
end
else begin
h_count <= h_count + 1'b1;
end
end
end
always @(posedge clk or negedge rst_n)
begin
if (!rst_n) begin
h_sync       <= 1'b1;
v_sync       <= 1'b1;
video_active <= 1'b0;
end
else begin
h_sync       <= ~((h_count >= (H_ACTIVE + H_FRONT)) && (h_count < (H_ACTIVE + H_FRONT + H_SYNC_P)));
v_sync       <= ~((v_count >= (V_ACTIVE + V_FRONT)) && (v_count < (V_ACTIVE + V_FRONT + V_SYNC_P)));
video_active <= (h_count < H_ACTIVE) && (v_count < V_ACTIVE);
end
end
localparam BLOCK_SIZE   = 3;
localparam BOARD_X_OFF  = 43;
localparam BOARD_Y_OFF  = 2;
wire [6:0] pixel_x = h_count[6:0];
wire [5:0] pixel_y = v_count[5:0];
wire [3:0] grid_x = (pixel_x - BOARD_X_OFF) / BLOCK_SIZE;
wire [4:0] grid_y = (pixel_y - BOARD_Y_OFF) / BLOCK_SIZE;
wire [1:0] block_inner_x = grid_x - current_block_x;
wire [1:0] block_inner_y = grid_y - current_block_y;
always @(posedge clk or negedge rst_n)
begin
if (!rst_n)
begin
pixel_data <= 1'b0;
end
else if (video_active)
begin
if (pixel_x >= (BOARD_X_OFF - 1) && pixel_x <= (BOARD_X_OFF + 42) &&
pixel_y >= (BOARD_Y_OFF - 1) && pixel_y <= (BOARD_Y_OFF + 60))
begin
if (pixel_x == (BOARD_X_OFF - 1) || pixel_x == (BOARD_X_OFF + 42) ||
pixel_y == (BOARD_Y_OFF - 1) || pixel_y == (BOARD_Y_OFF + 60))
begin
pixel_data <= 1'b1;
end
else
begin
if (grid_x < 14 && grid_y < 20 && board[grid_x][grid_y])
begin
pixel_data <= 1'b1;
end
else if (grid_x >= current_block_x && grid_x < (current_block_x + 4) &&
grid_y >= current_block_y && grid_y < (current_block_y + 4))
begin
pixel_data <= tetromino[block_inner_x][block_inner_y];
end
else
begin
pixel_data <= 1'b0;
end
end
end
else if (pixel_x == 7'd64)
begin
pixel_data <= pixel_y;
end
else
begin
pixel_data <= 1'b0;
end
end
else
begin
pixel_data <= 1'b0;
end
end
